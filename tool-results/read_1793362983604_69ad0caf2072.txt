    24→// Block-style English body paragraph: left-aligned, no first-line indent.
    25→function para(text, opts = {}) {
    26→  const { children, ...rest } = opts;
    27→  return new Paragraph({
    28→    alignment: AlignmentType.LEFT,
    29→    spacing: { line: 312, after: 160 },
    30→    children: children || [run(text, rest.run || {})],
    31→    ...(rest.paragraph || {}),
    32→  });
    33→}

    68→function cell(text, { bold = false, width, fill, size = SMALL, align = AlignmentType.LEFT } = {}) {
    69→  const lines = Array.isArray(text) ? text : [text];
    70→  return new TableCell({
    71→    margins: CELL_MARGINS,

    75→      alignment: align,
    76→      spacing: { line: 288 },
    77→      children: [new TextRun({ text: t, bold, size, color: BLACK, font: FONT })],
    78→    })),
    79→  });
    80→}

    92→  });
